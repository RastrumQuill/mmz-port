# motioncheck.py scenarios (MMZ1). The part before the first [name] is the
# game-specific config: route commands run after the warp, memory to log
# (each change is listed with its frame, to tie a pop-in to its cause),
# view sizes to run at.
views x2,x3,1120x480,900x520
settle 90
# the window's default the user plays with: rooms not fitted (the view stays at the size asked for)
env MMZ_FIT_ROOMS=0
# the HUD layer: drawn at the view's edges (camera check: no parallax judged)
hud 0
# (headless runs size the view as the window does: the fit, fit_small_rooms)
setup 10 cheat invincible 1
setup 10 cheat noenemies 1
watch mission 0x0202A574:1
watch stage 0x02022B60:2
watch area 0x0200520C:2
watch variants 0x02005208:4
watch checkpoint 0x02022AEC:1
# static checks: the rooms of each area/variant (chunk -> room, layer map level steps <= 1), keyed by the
# watched area and its variant (2 bits per area)
regions python3 port/tools/regionaudit.py ${MMZ_ROM:-baserom.gba} --rooms-json
regionkey "%d:%d" % (area & 0x7F, (variants >> ((area & 0x7F) * 2)) & 3)
static black_mean 0.08 black_max 0.30 foreign_max 0.01
